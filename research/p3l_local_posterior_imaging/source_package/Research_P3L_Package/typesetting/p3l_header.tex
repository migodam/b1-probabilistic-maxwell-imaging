\newcount\PThreeDisplayNo
\newcommand{\PThreeDisplay}[1]{%
  \global\advance\PThreeDisplayNo by 1%
  \[\sbox{\PThreeMathBox}{$\displaystyle #1$}%
  \ifdim\wd\PThreeMathBox>\linewidth
    \typeout{P3L-MATH-FIT \the\PThreeDisplayNo: natural=\the\wd\PThreeMathBox; available=\the\linewidth}%
    \resizebox{\linewidth}{!}{\usebox{\PThreeMathBox}}%
  \else\usebox{\PThreeMathBox}\fi\]}
